║    'assets/sounds/chanson2.mp3',                     ║
║  ];                                                  ║
║  const PLAYLIST_NAMES = [                            ║
║    'Ma Chanson 1',                                   ║
║    'Ma Chanson 2',                                   ║
║  ];                                                  ║

║  OU : utilisez le bouton "+ MP3" dans le jeu         ║
